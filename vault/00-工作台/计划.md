## 今日安排

```dataview
TABLE priority AS "优先级", status AS "状态", ref AS "关联"
FROM "09-计划/任务"
WHERE type = "task" AND date = date(today) AND status != "done"
SORT priority ASC
```

## 待安排与逾期

```dataviewjs
try {
  const tasks = dv.pages('"09-计划/任务"').where(p => p.type === "task" && p.status !== "done");
  const today = dv.date("today");
  const floating = tasks.where(p => !p.date);
  const overdue = tasks.where(p => p.date && p.date < today);
  dv.header(3, "待安排（" + floating.length + "）");
  dv.list(floating.length ? floating.map(p => p.file.link) : ["—"]);
  dv.header(3, "已逾期（" + overdue.length + "）");
  dv.list(overdue.length ? overdue.sort(p => p.date).map(p => p.file.link + " ｜ " + p.date.toFormat("yyyy-MM-dd")) : ["—"]);
} catch (e) {
  dv.paragraph("待安排渲染失败：" + e.message);
}
```
