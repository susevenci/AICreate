## 四类信号

```dataviewjs
try {
  const cfg = dv.page("00-工作台/_配置");
  const lowFan = Number(cfg?.lowFanThreshold ?? 10000);
  const eng = Number(cfg?.engagementThreshold ?? 0.05);
  const multi = Number(cfg?.outlierMultiplier ?? 2);
  const sources = dv.pages('"03-对标/作品"').where(p => p.type === "source");
  const bloggers = dv.pages('"03-对标/博主"').where(p => p.type === "blogger");
  const engagement = (p) => {
    const views = Number(p.views) || 0;
    const acts = (Number(p.likes) || 0) + (Number(p.comments) || 0) + (Number(p.shares) || 0);
    return views > 0 ? acts / views : 0;
  };
  dv.table(
    ["信号", "条数", "口径"],
    [
      ["监控博主", bloggers.length + " 位", "blogger 笔记数"],
      ["低粉爆款", sources.where(p => (Number(p.followers) || 0) < lowFan && engagement(p) > eng).length + " 条", "粉丝 < " + lowFan + " 且互动率 > " + eng],
      ["爆款内容异常值", sources.where(p => (Number(p.views) || 0) > (Number(p.baseline) || 0) * multi).length + " 条", "播放 > 基线 × " + multi],
    ]
  );
} catch (e) {
  dv.paragraph("对标统计渲染失败：" + e.message);
}
```
