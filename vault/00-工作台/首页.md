## 本月经营

```dataviewjs
try {
  const cfg = dv.page("00-工作台/_配置");
  if (!cfg || !cfg.name) {
    dv.paragraph("> 尚未配置工作台名称。打开 Web 端会引导完成初始化。");
  }
  const ym = dv.date("today").toFormat("yyyy-MM");
  const led = dv.pages('"10-收支"').where(p => p.type === "ledger" && p.date && p.date.toFormat("yyyy-MM") === ym);
  const sum = (kind) => led.where(p => p.kind === kind).reduce((s, p) => s + (Number(p.amount) || 0), 0);
  const income = sum("income");
  const expense = sum("expense");
  dv.table(
    ["指标", "数值", "口径"],
    [
      ["本月收入", "¥" + income.toFixed(2), "按流水金额合计，非笔数"],
      ["本月支出", "¥" + expense.toFixed(2), "按流水金额合计"],
      ["本月净额", "¥" + (income - expense).toFixed(2), "收入 - 支出"],
    ]
  );
  const missing = dv.pages('"10-收支"').where(p => p.type === "ledger" && !p.date).length;
  if (missing > 0) dv.paragraph(`> 数据质量：${missing} 条流水缺少日期，未计入月度统计。`);
} catch (e) {
  dv.paragraph("首页统计渲染失败：" + e.message);
}
```
